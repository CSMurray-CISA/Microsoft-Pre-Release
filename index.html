// =============================== // Clean CVRF → Parsed Vulnerabilities // =============================== function safeStr(v) { return v == null ? "" : (typeof v === "string" ? v : JSON.stringify(v)); } function firstText(...vals) { for (const v of vals) { const s = safeStr(v).trim(); if (s) return s; } return ""; } function pickCVSS(scoreSets) { if (!Array.isArray(scoreSets) || !scoreSets.length) return null; const ranked = scoreSets.map(s => { const ver = safeStr(s.Version || s.version); let base = parseFloat(s.BaseScore || s.baseScore); if (isNaN(base)) base = null; const pri = { "3.1": 3, "3.0": 2, "2.0": 1 }[ver] || 0; return { pri, base }; }).sort((a, b) => (b.pri - a.pri) || ((b.base ?? 0) - (a.base ?? 0))); return ranked[0].base; } function severity(score) { if (score == null || isNaN(score)) return "UNKNOWN"; if (score >= 9.0) return "CRITICAL"; if (score >= 7.0) return "HIGH"; if (score >= 4.0) return "MEDIUM"; if (score >= 0.0) return "LOW"; return "UNKNOWN"; } function extractVuln(v) { // --- CVE --- const cve = Array.isArray(v.CVE) ? v.CVE[0] : v.CVE || ""; // --- Description --- let desc = safeStr(v.Title?.Value); if (!desc) { for (const n of (v.Notes || [])) { const t = safeStr(n.Title).toLowerCase(); if (["summary", "description", "title"].includes(t)) { desc = firstText(n.Value, n.Text); break; } } } // --- CVSS + Severity --- const cvssBase = pickCVSS(v.CVSSScoreSets || []); const sev = severity(cvssBase); // --- SSVC --- let ssvc = ""; for (const n of (v.Notes || [])) { const t = safeStr(n.Title).toLowerCase(); if (t.includes("ssvc")) { ssvc = firstText(n.Text, n.Value); break; } } // --- Impact / Automatable --- let impact = ""; let autom = ""; for (const th of (v.Threats || [])) { if (!th || typeof th !== "object") continue; const type = safeStr(th.Type).toLowerCase(); if (type.includes("impact") && Array.isArray(th.Details)) { impact = firstText(...th.Details.map(d => d?.Value)); } if (type.includes("automatable") && Array.isArray(th.Details)) { const raw = firstText(...th.Details.map(d => d?.Value)).toLowerCase(); autom = raw.includes("yes") ? "Yes" : raw.includes("no") ? "No" : "Automatable"; } } return { "CVE-ID": cve, "Description": desc, "SSVC": ssvc, "CVSS": cvssBase == null ? "" : cvssBase.toFixed(1), "Severity": sev, "Impact": impact, "Automatable": autom, // These are kept blank intentionally so your HTML table still has columns "Internet Exposed": "", "Forensic Triage Required": "" }; } function parseCVRF(json) { const vulns = Array.isArray(json?.Vulnerability) ? json.Vulnerability : Array.isArray(json?.Document?.Vulnerability) ? json.Document.Vulnerability : []; return vulns .map(extractVuln) .sort((a, b) => parseFloat(b.CVSS || 0) - parseFloat(a.CVSS || 0)); } // =============================== // HOW TO USE: // const parsed = parseCVRF(yourJsonObject); // console.log(parsed); // => array of vulnerabilities with all metrics // ===============================
